# Way2API - IP Geolocation Lookup
# POST https://app.way2api.com/api/v1/ip/geolocation
# Docs: https://app.way2api.com/documentation/ip-geolocation
#
# Get your API key at https://app.way2api.com/register
# The key may be sent as "Authorization: Bearer <key>" or "X-API-Key: <key>".

curl -X POST "https://app.way2api.com/api/v1/ip/geolocation" \
  -H "Authorization: Bearer YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "ip_address": "8.8.8.8"
  }'
