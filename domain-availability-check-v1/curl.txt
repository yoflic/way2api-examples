# Way2API - Domain Availability Check
# POST https://app.way2api.com/api/v1/domain/check
# Docs: https://app.way2api.com/documentation/domain-availability-check-v1
#
# Get your API key at https://app.way2api.com/register
# The key may be sent as "Authorization: Bearer <key>" or "X-API-Key: <key>".

curl -X POST "https://app.way2api.com/api/v1/domain/check" \
  -H "Authorization: Bearer YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "domain": "example.com"
  }'
