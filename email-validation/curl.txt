# Way2API - Email Validation
# POST https://app.way2api.com/api/v1/email/validate
# Docs: https://app.way2api.com/documentation/email-validation
#
# Get your API key at https://app.way2api.com/register
# The key may be sent as "Authorization: Bearer <key>" or "X-API-Key: <key>".

curl -X POST "https://app.way2api.com/api/v1/email/validate" \
  -H "Authorization: Bearer YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "email": "emma.thompson@example.com"
  }'
